## 7. Pseudocode

```text
aggregate(A, B):
  K   ← E_A[J_St] + E_B[J_St]                      # không ghép
  EΔ  ← E_A[δ] + E_B[δ];  EΔ̂ ← E_A[δ̂] + E_B[δ̂]     # không ghép
  a00 ← [EΔ̂ < 0];  a01 ← [EΔ < 0]                   # một action cho cả ô
  (p_sw, −EΔ⁻) ← cut_sum(key=δ,  val=δ)             # oracle
  sw10         ← cut_sum(key=δ̂, val=δ)              # quyết theo δ̂, chấm bằng δ
  gộp mode trong cùng backlog → (P, δ̄, δ̂̄) mỗi phía
  swcb         ← cut_sum(key=δ̂̄, val=δ̄)             # D22
  L ← {00: K+a00·EΔ, 01: K+a01·EΔ, 10: K+sw10, 11: K−EΔ⁻, cb0: K+swcb}
```

### 8.1. `factorial_core.py`: tầng sổ sách

```python
"""Tầng sổ sách của engine exact (L3.2): ghép hai bảng một phía thành bốn L.

Scientific purpose: tính L00, L01, L10, L11, L_cb0 (D22) và sáu contrast D19
    từ hai bảng tách được, KHÔNG dựng tích D_A x D_B.
Used by: evaluate_cell (A01, A04, A05-T); test golden Phase 1, E2, E7.
Metric: packet·s. Bản này dùng Fraction (chính xác) cho toy và golden.
"""
from bisect import bisect_left
from dataclasses import dataclass
from fractions import Fraction as F


@dataclass(frozen=True)
class Side:
    """Bảng một phía. Mỗi dòng là một state x_l = (q_l, m_l) của queue l."""
    p: tuple      # D_l(x_l)
    d: tuple      # δ_l: thành phần gap thật (M1)
    dh: tuple     # δ̂_l: thành phần gap theo M0
    jst: tuple    # J_l(Stay; x_l) thật
    group: tuple  # mã backlog q_l, dùng cho D22 (giấu mode)

    def check(self, which):
        n = len(self.p)
        if not all(len(v) == n for v in (self.d, self.dh, self.jst, self.group)):
            raise ValueError('cột lệch độ dài')
        if any(x <= 0 for x in self.p) or sum(self.p) != 1:
            raise ValueError('D_l phải dương trên support và cộng bằng 1')
        sign = (lambda v: v >= 0) if which == 'A' else (lambda v: v <= 0)
        if not all(sign(v) for v in self.d + self.dh):
            raise ValueError(f'vi phạm dấu δ_{which} (Part 2 §7.3)')


def _cut_sum(A_key, A_val, A_p, B_key, B_val, B_p):
    """Σ_a p_a Σ_{b: keyA+keyB<0} p_b (valA+valB), và khối lượng tương ứng.

    Sắp B theo KEY (cái dùng để quyết định), cộng dồn VAL (cái dùng để chấm).
    Bất đẳng thức chặt: hòa thì Stay (D16).
    """
    order = sorted(range(len(B_key)), key=lambda i: B_key[i])
    keys = [B_key[i] for i in order]
    cum_p, cum_pv = [F(0)], [F(0)]
    for i in order:
        cum_p.append(cum_p[-1] + B_p[i])
        cum_pv.append(cum_pv[-1] + B_p[i] * B_val[i])
    mass = value = F(0)
    for ka, va, pa in zip(A_key, A_val, A_p):
        k = bisect_left(keys, -ka)              # số phần tử < -ka
        mass += pa * cum_p[k]
        value += pa * (va * cum_p[k] + cum_pv[k])
    return mass, value


def _by_backlog(s):
    """Gộp mode trong cùng backlog: posterior D(m|q), trung bình δ và δ̂ (D22, D10)."""
    acc = {}
    for p, d, dh, g in zip(s.p, s.d, s.dh, s.group):
        P, Sd, Sdh = acc.get(g, (F(0), F(0), F(0)))
        acc[g] = (P + p, Sd + p * d, Sdh + p * dh)
    P = tuple(v[0] for v in acc.values())
    return P, tuple(v[1] / v[0] for v in acc.values()), tuple(v[2] / v[0] for v in acc.values())


def aggregate(A, B):
    A.check('A'); B.check('B')
    E = lambda s, col: sum(p * v for p, v in zip(s.p, col))
    K = E(A, A.jst) + E(B, B.jst)                  # E_D J(Stay): không cần ghép
    Ed, Edh = E(A, A.d) + E(B, B.d), E(A, A.dh) + E(B, B.dh)
    a00, a01 = int(Edh < 0), int(Ed < 0)           # S0: MỘT action cho cả ô (T01)
    p_sw, neg = _cut_sum(A.d, A.d, A.p, B.d, B.d, B.p)     # E[Δ·1(Δ<0)] = -EΔ⁻
    _, sw10 = _cut_sum(A.dh, A.d, A.p, B.dh, B.d, B.p)     # quyết theo δ̂, chấm bằng δ
    PA, dA, dhA = _by_backlog(A); PB, dB, dhB = _by_backlog(B)
    _, swcb = _cut_sum(dhA, dA, PA, dhB, dB, PB)           # D22: biết q, giấu mode
    L = {'00': K + a00 * Ed, '01': K + a01 * Ed, '10': K + sw10, '11': K + neg, 'cb0': K + swcb}
    G = dict(G_S=L['00'] - L['10'], G_M=L['00'] - L['01'], G_SM=L['00'] - L['11'],
             Gamma=L['10'] + L['01'] - L['00'] - L['11'],
             G_S_M1=L['01'] - L['11'], G_M_S1=L['10'] - L['11'],
             F=L['00'] - L['cb0'], V=L['cb0'] - L['10'])
    return dict(L=L, G=G, a00=a00, a01=a01, E_delta=Ed, E_delta_hat=Edh, bias=Edh - Ed,
                E_pos=Ed - neg, E_neg=-neg, p_sw=p_sw, C_ref=L['11'])
```

**Đọc theo khối:**
- **`Side.check`.** Nếu vật lý đưa lên một $\delta_A$ âm, ta dừng ngay *trước* khi ghép. Bug vật lý không được lặn vào tầng sổ sách.
- **`_cut_sum`.** Đây là toàn bộ Part 3 §7.3 trong khoảng 15 dòng. Một hàm duy nhất phục vụ ba việc: $p_{\text{sw}}$ và $L_{11}$ (khi key = val = $\delta$), $L_{10}$ (key = $\hat\delta$, val = $\delta$), và D22 (bảng đã gộp mode). Ít code thì ít chỗ cho bug, nhưng một bug ở đây sẽ lan ra mọi nơi. Vì vậy hàm này cần được kiểm bằng liệt kê.
- **`_by_backlog`.** Lấy trung bình $\hat\delta$ theo mode là hợp lệ **vì có D10**. Nếu $M_0$ rollout từ state trung bình, dòng này sẽ sai.
- **`aggregate`.** $a_{00}$ và $a_{01}$ chỉ đọc *trung bình*. Đó là quyền thông tin của $S_0$, được viết thẳng vào code.

### 8.2. `engine_checker.py`: checker độc lập

```python
"""Checker độc lập: liệt kê mọi cặp (x_A, x_B). KHÔNG import factorial_core."""
from fractions import Fraction as F
from itertools import product


def brute(A, B):
    rows = []
    for (pa, da, ha, ja, ga), (pb, db, hb, jb, gb) in product(
            zip(A.p, A.d, A.dh, A.jst, A.group), zip(B.p, B.d, B.dh, B.jst, B.group)):
        rows.append(dict(p=pa * pb, st=ja + jb, sw=ja + jb + da + db, dh=ha + hb, gA=ga, gB=gb))
    Ed = sum(r['p'] * (r['sw'] - r['st']) for r in rows)
    Edh = sum(r['p'] * r['dh'] for r in rows)
    pick = lambda r, switch: r['sw'] if switch else r['st']
    L00 = sum(r['p'] * pick(r, Edh < 0) for r in rows)
    L01 = sum(r['p'] * pick(r, Ed < 0) for r in rows)
    L10 = sum(r['p'] * pick(r, r['dh'] < 0) for r in rows)
    L11 = sum(r['p'] * min(r['st'], r['sw']) for r in rows)
    groups = {}
    for r in rows:                                   # D22 theo định nghĩa: posterior trên nhóm (qA,qB)
        g = groups.setdefault((r['gA'], r['gB']), [])
        g.append(r)
    Lcb = F(0)
    for g in groups.values():
        P = sum(r['p'] for r in g)
        switch = sum(r['p'] * r['dh'] for r in g) / P < 0
        Lcb += sum(r['p'] * pick(r, switch) for r in g)
    return dict(L={'00': L00, '01': L01, '10': L10, '11': L11, 'cb0': Lcb},
                p_sw=sum(r['p'] for r in rows if r['sw'] < r['st']), rows=rows)
```

**Vì sao checker phải viết khác hẳn.** Checker đi theo *định nghĩa*, không theo *thuật toán*:
- dựng sổ cái đầy đủ;
- chọn action bằng cách so `sw` với `st`;
- D22 lấy posterior trên **nhóm cặp** $(q_A,q_B)$.

Core thì đi theo con đường hoàn toàn khác: sắp xếp, cộng dồn, gộp từng phía. Hai con đường chung một bug là điều rất khó xảy ra. Đây là bài học trực tiếp từ L1.5.

### 8.3. `toy_physics.py`: vật lý cho toy, chỉ dùng hàm có sẵn của repo

```python
"""Tầng vật lý cho plant TOY, chỉ dùng hàm tham chiếu Fraction đã có trong repo.

Scientific purpose: dựng bảng Side từ một plant nhỏ để engine tái lập golden Phase 1.
Không dùng cho ô lưới (ô lưới đi qua cells.py + kernel native).
"""
from fractions import Fraction as F
from smr.exact import value_backward
from smr.idle_gap import weighted_idle, mean_weighted_idle
from experiments.A00_engine.factorial_core import Side


def toy_side(states, rates, P, c, u_stay, u_switch, H, dt=F(1)):
    """states: list of (prob, q, mode). Trả Side và (J_Sw thật) để kiểm chéo."""
    cols = dict(p=[], d=[], dh=[], jst=[], group=[]); jsw = []
    for p, q, m in states:
        w1 = {a: weighted_idle(q, m, rates, x, c, H, P, dt) for a, x in (('St', u_stay), ('Sw', u_switch))}
        w0 = {a: mean_weighted_idle(q, m, rates, x, c, H, P, dt) for a, x in (('St', u_stay), ('Sw', u_switch))}
        if not all(w0[a] <= w1[a] for a in w1):
            raise AssertionError('Jensen: W^M0 <= W^M1 bị vi phạm')
        cols['p'].append(F(p)); cols['d'].append(w1['Sw'] - w1['St']); cols['dh'].append(w0['Sw'] - w0['St'])
        cols['jst'].append(value_backward(u_stay, H, c, rates, P, [q], dt)[q, m]); cols['group'].append(q)
        jsw.append(value_backward(u_switch, H, c, rates, P, [q], dt)[q, m])
    return Side(**{k: tuple(v) for k, v in cols.items()}), tuple(jsw)
```

Hàm này **không viết vật lý mới**. Nó chỉ ghép ba hàm tham chiếu đã được test ở Phase 1:
- `value_backward` (cost cộng dồn);
- `weighted_idle` (capacity bỏ phí theo $M_1$);
- `mean_weighted_idle` (đường trung bình $M_0$).

$J(\text{Sw})$ được tính riêng để làm kiểm chéo loại 3.

```python
import random

def side(p, d, dh, jst=None, group=None):

def canonical():
    """Ví dụ chuẩn Phase 1: H=2, c=3, u=2, A nền 1, B on/off 5/0, keep 1/2, D đều trên 4 state."""
    keep = {'on': {'on': h, 'off': h}, 'off': {'on': h, 'off': h}}

    return A, B, jswA, jswB

E7 = (side([h, h], [1, 3], ['1/2', '3/2'], group=[1, 4]),

    r = aggregate(*canonical()[:2])
```

## 10. Bản float có khoảng: hiện thực hóa đề xuất §8b của Part 3

Ô thật dùng float, nên mỗi key có sai số. Ta cắt hai lần: một lần ở "chắc chắn Switch", một lần ở "có thể Switch". Phần nằm giữa hai lần cắt là phần chưa phân giải, và ta tính được cận ảnh hưởng của nó:

```python
import numpy as np

def cut_with_bound(A_key, A_val, A_p, B_key, B_val, B_p, e_key):
    """Như _cut_sum nhưng key có sai số ±e_key (đã cộng cả hai phía).

    Trả: value (cắt tại giá trị danh nghĩa), khối lượng chưa phân giải,
    và cận ảnh hưởng Σ p·|Δ| trên các cặp chưa phân giải (Part 3 §8b).
    """
    o = np.argsort(B_key, kind='stable'); keys = B_key[o]
    cp  = np.r_[0.0, np.cumsum(B_p[o])]
    cpv = np.r_[0.0, np.cumsum(B_p[o] * B_val[o])]
    cpa = np.r_[0.0, np.cumsum(B_p[o] * np.abs(B_val[o]))]
    k  = np.searchsorted(keys, -A_key,         side='left')   # danh nghĩa
    lo = np.searchsorted(keys, -A_key - e_key, side='left')   # chắc chắn Switch
    hi = np.searchsorted(keys, -A_key + e_key, side='left')   # có thể Switch
    value = np.sum(A_p * (A_val * cp[k] + cpv[k]))
    band = cp[hi] - cp[lo]
    unresolved = np.sum(A_p * band)
    impact = np.sum(A_p * (np.abs(A_val) * band + (cpa[hi] - cpa[lo])))
    return value, unresolved, impact
```

**Cận ảnh hưởng.** Nhờ bất đẳng thức tam giác, $\lvert\delta_A+\delta_B\rvert\le\lvert\delta_A\rvert+\lvert\delta_B\rvert$. Vì vậy chỉ cần thêm **một** mảng cộng dồn $p\lvert\delta_B\rvert$.

**Đã kiểm.** Trên 2000 toy ngẫu nhiên, mình xê dịch key trong khoảng ±e. Độ lệch của `value` không bao giờ vượt `impact`, ngoại trừ phần làm tròn cỡ 4·10⁻¹⁶. Bản này dùng cho $L_{10}$ và $L_{\text{cb}0}$. Với $L_{11}$ thì không cần, vì $\min$ liên tục (Part 3 §7.7).

## 12. Guard: hiện thực hóa thứ tự commit của L3.1, Part 2

```python
GRID = json.load(open('notes/design/grid_v1_2.json'))
GRID_IDS = {c['id'] for k in ('main', 'heldout', 'controls', 'probes') for c in GRID[k]}
HELDOUT_IDS = {c['id'] for c in GRID['heldout']}
CARD = 'notes/lessons/L3.3_prediction_card.md'      # tên tạm để minh họa

def _committed(path):
    head = subprocess.run(['git', 'log', '-1', '--format=%H', '--', path],
                          capture_output=True, text=True, check=True).stdout.strip()
    dirty = subprocess.run(['git', 'status', '--porcelain', '--', path],
                           capture_output=True, text=True, check=True).stdout.strip()
    if not head or dirty:
        raise PermissionError(f'{path} chưa commit hoặc đang sửa dở')
    return head

def guard(cell_id, *, allow_grid=False, allow_heldout=False):
    if cell_id not in GRID_IDS:
        return None                                   # toy: tự do
    if not allow_grid:
        raise PermissionError(f'{cell_id} là ô lưới: cần --allow-grid (chỉ runner A01/A04/A05-T)')
    if cell_id in HELDOUT_IDS and not allow_heldout:
        raise PermissionError(f'{cell_id} là held-out: chỉ A05-T, sau khi A05-F đã commit')
    return _committed(CARD)                           # ghi hash này vào mọi output
```

Hai chỗ khác với plan, kèm lý do:
- **Guard theo id, không theo `role`.** P01 và P02 không có trường `role`, nên guard theo role sẽ để lọt hai đầu dò.
- **Held-out có cờ riêng.** Runner của A01 cần `--allow-grid`, nhưng không bao giờ được chạm R01–R12.
